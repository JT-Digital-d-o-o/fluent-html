const confirm = modal("delete-project");

const DeleteDialog = ({ id, name }: Project) => (
  <dialog class="w-full max-w-sm rounded-card bg-surface p-6 shadow-lg backdrop:bg-overlay/40" {...confirm.root}>
    <h2 class="text-lg font-semibold">Delete project?</h2>
    <p class="mt-2 text-sm text-text-dim">{`"${name}" and its exports will be removed. This cannot be undone.`}</p>
    <div class="mt-6 flex justify-end gap-3">
      <button class={quietButton} {...confirm.close}>Cancel</button>
      <button class={dangerButton} {...projects.remove({ id })}>Delete project</button>
    </div>
  </dialog>
);

export const ProjectDetailPage = (project: Project) => (
  <Layout title={project.name}>
    <a class="text-sm text-text-dim hover:underline" {...projects.list()}>Projects</a>
    <div class="mt-2 flex items-start justify-between gap-4">
      <div>
        <h1 class="text-2xl font-semibold">{project.name}</h1>
        <p class="mt-1 text-text-dim">{project.description}</p>
      </div>
      <button class={dangerButton} {...confirm.open}>Delete</button>
    </div>
    <dl class="mt-6 grid grid-cols-[8rem_1fr] gap-y-3 text-sm">
      <dt class="text-text-dim">Status</dt>
      <dd>{StatusPill(project.status)}</dd>
      <dt class="text-text-dim">Export</dt>
      <dd>{JobBadge(project)}</dd>
    </dl>
    {DeleteDialog(project)}
  </Layout>
);
