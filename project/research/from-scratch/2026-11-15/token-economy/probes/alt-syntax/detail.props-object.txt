const confirm = modal("delete-project");

const DeleteDialog = ({ id, name }: Project) =>
  dialog(
    { class: "w-full max-w-sm rounded-card bg-surface p-6 shadow-lg backdrop:bg-overlay/40", ...confirm.root },
    h2({ class: "text-lg font-semibold" }, "Delete project?"),
    p({ class: "mt-2 text-sm text-text-dim" }, `"${name}" and its exports will be removed. This cannot be undone.`),
    div(
      { class: "mt-6 flex justify-end gap-3" },
      button({ class: quietButton, ...confirm.close }, "Cancel"),
      button({ class: dangerButton, ...projects.remove({ id }) }, "Delete project"),
    ),
  );

export const ProjectDetailPage = (project: Project) =>
  Layout(
    { title: project.name },
    a({ class: "text-sm text-text-dim hover:underline", ...projects.list() }, "Projects"),
    div(
      { class: "mt-2 flex items-start justify-between gap-4" },
      div(h1({ class: "text-2xl font-semibold" }, project.name), p({ class: "mt-1 text-text-dim" }, project.description)),
      button({ class: dangerButton, ...confirm.open }, "Delete"),
    ),
    dl(
      { class: "mt-6 grid grid-cols-[8rem_1fr] gap-y-3 text-sm" },
      dt({ class: "text-text-dim" }, "Status"),
      dd(StatusPill(project.status)),
      dt({ class: "text-text-dim" }, "Export"),
      dd(JobBadge(project)),
    ),
    DeleteDialog(project),
  );
